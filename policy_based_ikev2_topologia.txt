# Topología VPN Site-to-Site Policy-Based con IKEv2
# Estructura:
# R1 ---- ISP(R5) ---- R2
#  |                   |
# SW-LEFT            SW-RIGHT
#  |  |               |   |
# VLAN10 VLAN20     VLAN30 VLAN40

############################################################
# 1) SWITCH IZQUIERDO (SW-LEFT)
############################################################
enable
conf t
hostname SW-LEFT

vlan 10
 name DEVOPS

vlan 20
 name PENTESTING

interface e0/0
 switchport mode trunk
 no shutdown

interface e2/1
 switchport mode access
 switchport access vlan 10
 no shutdown

interface e3/1
 switchport mode access
 switchport access vlan 20
 no shutdown

end
write memory

############################################################
# 2) SWITCH DERECHO (SW-RIGHT)
############################################################
enable
conf t
hostname SW-RIGHT

vlan 30
 name MARKETING

vlan 40
 name SERVICE_CLIENT

interface e0/0
 switchport mode trunk
 no shutdown

interface e2/1
 switchport mode access
 switchport access vlan 30
 no shutdown

interface e3/1
 switchport mode access
 switchport access vlan 40
 no shutdown

end
write memory

############################################################
# 3) ROUTER IZQUIERDO (R1 / Enterprise1) - IKEv2
############################################################
enable
conf t
hostname Enterprise1

interface e0/0
 ip address 200.25.41.1 255.255.255.252
 no shutdown

interface e0/1
 no ip address
 no shutdown

interface e0/1.10
 encapsulation dot1Q 10
 ip address 192.168.10.1 255.255.255.0

interface e0/1.20
 encapsulation dot1Q 20
 ip address 192.168.20.1 255.255.255.0

ip dhcp excluded-address 192.168.10.1 192.168.10.10
ip dhcp excluded-address 192.168.20.1 192.168.20.10

ip dhcp pool DEVOPS
 network 192.168.10.0 255.255.255.0
 default-router 192.168.10.1
 dns-server 8.8.8.8

ip dhcp pool PENTESTING
 network 192.168.20.0 255.255.255.0
 default-router 192.168.20.1
 dns-server 8.8.8.8

router ospf 1
 network 200.25.41.0 0.0.0.3 area 0
 network 192.168.10.0 0.0.0.255 area 0
 network 192.168.20.0 0.0.0.255 area 0

crypto ikev2 proposal IKEV2-PROP
 encryption aes-cbc-256
 integrity sha256
 group 14

crypto ikev2 policy IKEV2-POLICY
 proposal IKEV2-PROP

crypto ikev2 keyring VPN-KEYRING
 peer R2
  address 200.41.25.1
  pre-shared-key angel

crypto ikev2 profile VPN-PROFILE
 match identity remote address 200.41.25.1 255.255.255.255
 authentication remote pre-share
 authentication local pre-share
 keyring local VPN-KEYRING

crypto ipsec transform-set VPN-SET esp-aes esp-sha-hmac
 mode tunnel

access-list 110 permit ip 192.168.10.0 0.0.0.255 192.168.30.0 0.0.0.255
access-list 110 permit ip 192.168.10.0 0.0.0.255 192.168.40.0 0.0.0.255
access-list 110 permit ip 192.168.20.0 0.0.0.255 192.168.30.0 0.0.0.255
access-list 110 permit ip 192.168.20.0 0.0.0.255 192.168.40.0 0.0.0.255

crypto map VPN-IKEV2 10 ipsec-isakmp
 set peer 200.41.25.1
 set transform-set VPN-SET
 set ikev2-profile VPN-PROFILE
 match address 110

interface e0/0
 crypto map VPN-IKEV2

ip forward-protocol nd
no ip http server
no ip http secure-server

end
write memory
